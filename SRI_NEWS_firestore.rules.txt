rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isOwner() {
      return signedIn()
        && request.auth.uid == 'awKV4C8EnfN2B15hCYAcTkGy7n1l';
    }

    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isOwnerOrAdmin() {
      return isOwner() || isAdmin();
    }

    match /news/{newsId} {
      allow read: if true;
      allow create, delete: if isOwnerOrAdmin();
      allow update: if isOwnerOrAdmin()
        || (
          signedIn()
          && request.resource.data.diff(resource.data)
              .affectedKeys().hasOnly(['likes', 'likeCount'])
        );
    }

    match /news/{newsId}/likes/{userId} {
      allow read: if true;
      allow create: if signedIn()
        && userId == request.auth.uid
        && request.resource.data.userId == request.auth.uid;
      allow delete: if signedIn()
        && userId == request.auth.uid
        && resource.data.userId == request.auth.uid;
      allow update: if false;
    }

    match /news/{newsId}/comments/{commentId} {
      allow read: if true;
      allow create: if signedIn()
        && request.resource.data.userId == request.auth.uid;
      allow update, delete: if isOwnerOrAdmin()
        || (signedIn() && resource.data.userId == request.auth.uid);
    }

    match /users/{userId} {
      allow read: if isOwnerOrAdmin()
        || (signedIn() && userId == request.auth.uid);

      allow create: if isOwnerOrAdmin()
        || (signedIn() && userId == request.auth.uid);

      allow update: if isOwner()
        || (
          isAdmin()
          && userId != 'awKV4C8EnfN2B15hCYAcTkGy7n1l'
          && request.resource.data.role != 'owner'
        )
        || (
          signedIn()
          && userId == request.auth.uid
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
            'name',
            'email',
            'photoUrl',
            'displayName',
            'reporterStatus',
            'profileImage',
            'profilePhoto'
          ])
        );

      allow delete: if isOwner();
    }

    match /reporterApplications/{applicationId} {
      allow read: if isOwnerOrAdmin()
        || (
          signedIn()
          && (
            resource.data.userId == request.auth.uid
            || resource.data.reporterId == request.auth.uid
          )
        );

      allow create: if signedIn()
        && (
          request.resource.data.userId == request.auth.uid
          || request.resource.data.reporterId == request.auth.uid
        );

      allow update, delete: if isOwnerOrAdmin()
        || (
          signedIn()
          && (
            resource.data.userId == request.auth.uid
            || resource.data.reporterId == request.auth.uid
          )
        );
    }

    match /reporterPosts/{postId} {
      allow read: if isOwnerOrAdmin()
        || (
          signedIn()
          && resource.data.reporterId == request.auth.uid
        );

      allow create: if signedIn()
        && request.resource.data.reporterId == request.auth.uid
        && request.resource.data.status == 'pending';

      allow update: if isOwnerOrAdmin()
        || (
          signedIn()
          && resource.data.reporterId == request.auth.uid
          && resource.data.status == 'pending'
          && request.resource.data.reporterId == request.auth.uid
          && request.resource.data.status == 'pending'
        );

      allow delete: if isOwnerOrAdmin()
        || (
          signedIn()
          && resource.data.reporterId == request.auth.uid
          && resource.data.status == 'pending'
        );
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
